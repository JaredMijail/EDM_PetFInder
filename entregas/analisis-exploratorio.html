# --- formateo de cifras en español ---
fmt <- function(x, d = 2) {
  x <- as.numeric(x)
  if (all(abs(x - round(x)) < 1e-9)) {
    return(format(round(x), big.mark = ".", decimal.mark = ",", scientific = FALSE))
  }
  format(round(x, d), big.mark = ".", decimal.mark = ",", nsmall = d, scientific = FALSE)
}
ent <- function(x) format(as.integer(round(as.numeric(x))), big.mark = ".", decimal.mark = ",")
pct <- function(x, d = 1) gsub("\\.", ",", sprintf(paste0("%.", d, "f %%"), x))

# --- estadísticos ---
moda      <- function(x) { x <- x[!is.na(x)]; ux <- unique(x); ux[which.max(tabulate(match(x, ux)))] }
asimetria <- function(x) { x <- x[!is.na(x)]; n <- length(x); m <- mean(x); s <- sd(x); n/((n-1)*(n-2))*sum(((x-m)/s)^3) }
curtosis  <- function(x) { x <- x[!is.na(x)]; n <- length(x); m <- mean(x); s <- sd(x)
  n*(n+1)/((n-1)*(n-2)*(n-3))*sum(((x-m)/s)^4) - 3*(n-1)^2/((n-2)*(n-3)) }

esc_num <- function(x) {
  x <- as.numeric(x); v <- x[!is.na(x)]; q <- quantile(v, c(.25, .75)); iqr <- diff(q)
  z <- (v - mean(v)) / sd(v); li <- q[1] - 1.5*iqr; ls <- q[2] + 1.5*iqr
  list(n = length(v), na = sum(is.na(x)), media = mean(v), mediana = median(v), moda = as.numeric(moda(v)),
       min = min(v), max = max(v), q1 = q[1], q3 = q[2], iqr = iqr, rango = max(v) - min(v),
       var = var(v), sd = sd(v), cv = 100*sd(v)/mean(v), asim = asimetria(v), curt = curtosis(v),
       out_iqr = sum(v < li | v > ls), pct_out_iqr = 100*mean(v < li | v > ls),
       out_z = sum(abs(z) > 3), pct_out_z = 100*mean(abs(z) > 3), pct_ceros = 100*mean(v == 0),
       unicos = length(unique(v)))
}

esc_cat <- function(x) {
  v <- as.character(x); tb <- sort(table(v), decreasing = TRUE)
  list(n = length(v), niveles = length(tb), moda = names(tb)[1], n_moda = as.integer(tb[1]),
       pct_moda = 100*tb[1]/length(v), raras = sum(tb/length(v) < 0.01),
       pct_raras = 100*sum(tb[tb/length(v) < 0.01])/length(v),
       unicos = length(tb), unicidad = 100*length(tb)/length(v))
}

tabla_atipicos <- function(x, k = 3) {
  x <- as.numeric(x); v <- x[!is.na(x)]; q <- quantile(v, c(.25, .75)); iqr <- diff(q)
  z <- (v - mean(v))/sd(v); li <- q[1] - 1.5*iqr; ls <- q[2] + 1.5*iqr
  tibble::tibble(
    Criterio = c("Regla de 1,5 × IQR", paste0("Puntaje z (|z| > ", k, ")")),
    `Límite inferior` = c(fmt(li), fmt(mean(v) - k*sd(v))),
    `Límite superior` = c(fmt(ls), fmt(mean(v) + k*sd(v))),
    Atípicos = c(ent(sum(v < li | v > ls)), ent(sum(abs(z) > k))),
    `% del total` = c(pct(100*mean(v < li | v > ls)), pct(100*mean(abs(z) > k)))
  ) |> knitr::kable(align = c("l", "r", "r", "r", "r"))
}

tabla_frecuencias <- function(x, etiqueta = "Categoría", n_top = 10) {
  v <- as.character(x); tb <- sort(table(v), decreasing = TRUE); n <- length(tb); tot <- sum(tb)
  if (n > n_top) {
    tab <- tibble::tibble(
      Categoría = c(names(tb)[1:(n_top - 1)], sprintf("Otras (%d categorías)", n - n_top + 1)),
      Frecuencia = c(as.integer(tb[1:(n_top - 1)]), tot - sum(tb[1:(n_top - 1)])))
  } else {
    tab <- tibble::tibble(Categoría = names(tb), Frecuencia = as.integer(tb))
  }
  tab |>
    mutate(`Frecuencia relativa` = 100*Frecuencia/sum(Frecuencia),
           `Frecuencia acumulada` = cumsum(`Frecuencia relativa`)) |>
    mutate(Frecuencia = ent(Frecuencia),
           `Frecuencia relativa` = pct(`Frecuencia relativa`),
           `Frecuencia acumulada` = pct(`Frecuencia acumulada`)) |>
    knitr::kable(col.names = c(etiqueta, "Frecuencia absoluta", "Frecuencia relativa",
                               "Frecuencia acumulada"), align = c("l", "r", "r", "r"))
}

tema_u <- function() {
  theme_minimal(base_size = 11.5) +
    theme(plot.background = element_rect(fill = "white", colour = NA),
          panel.background = element_rect(fill = "white", colour = NA),
          plot.title = element_text(colour = "#1a3a5c", face = "bold", size = 11.5),
          axis.title = element_text(colour = "#4b5563", size = 10),
          panel.grid.minor = element_blank())
}

graficos_cuanti <- function(x, etiqueta, tipo = c("continuo", "conteo")) {
  tipo <- match.arg(tipo)
  d <- tibble::tibble(v = as.numeric(x))
  p1 <- ggplot(d, aes(v)) +
    geom_histogram(bins = 30, fill = "#2e7d9a", colour = "white", linewidth = 0.2) +
    labs(title = "Histograma", x = etiqueta, y = "Frecuencia") + tema_u()
  p2 <- ggplot(d, aes(y = v)) +
    geom_boxplot(fill = "#d97706", width = 0.32, outlier.colour = "#dc2626", outlier.alpha = 0.3) +
    labs(title = "Diagrama de caja", x = NULL, y = etiqueta) +
    theme(axis.text.x = element_blank()) + tema_u()
  p3 <- ggplot(d, aes(v)) +
    geom_density(fill = "#0f766e", alpha = 0.35, colour = "#0f766e") +
    labs(title = "Densidad", x = etiqueta, y = "Densidad") + tema_u()
  p4 <- if (tipo == "continuo") {
    ggplot(d, aes(sample = v)) +
      stat_qq(colour = "#1a3a5c", alpha = 0.4, size = 0.8) +
      stat_qq_line(colour = "#d97706") +
      labs(title = "QQ-plot (referencia normal)", x = "Cuantiles teóricos", y = "Cuantiles muestrales") + tema_u()
  } else {
    d |> count(v) |> slice_max(n, n = 8) |> mutate(v = factor(v)) |>
      ggplot(aes(x = v, y = n)) + geom_col(fill = "#1a3a5c") +
      labs(title = "Valores más frecuentes", x = etiqueta, y = "Frecuencia") + tema_u()
  }
  for (p in list(p1, p2, p3, p4)) print(p)
  invisible(NULL)
}

graficos_cuali <- function(x, etiqueta, pastel = FALSE, n_top = 10) {
  v <- as.character(x); tb <- sort(table(v), decreasing = TRUE); n <- length(tb); tot <- sum(tb)
  d <- if (n > n_top) {
    tibble::tibble(cat = c(names(tb)[1:(n_top - 1)], sprintf("Otras (%d)", n - n_top + 1)),
                   n = c(as.integer(tb[1:(n_top - 1)]), tot - sum(tb[1:(n_top - 1)])))
  } else tibble::tibble(cat = names(tb), n = as.integer(tb))
  d <- d |> mutate(cat = factor(cat, levels = rev(cat)),
                   pct = 100*n/sum(n))
  p1 <- ggplot(d, aes(x = pct, y = cat)) +
    geom_col(fill = "#2e7d9a", width = 0.7) +
    geom_text(aes(label = sprintf("%.1f %%", pct)), hjust = -0.15, size = 3, colour = "#1f2937") +
    scale_x_continuous(expand = expansion(mult = c(0, 0.22))) +
    labs(title = "Distribución de frecuencias", x = "Porcentaje de anuncios", y = etiqueta) + tema_u()
  if (!pastel) { print(p1); return(invisible(NULL)) }
  p2 <- ggplot(d, aes(x = "", y = n, fill = cat)) +
    geom_col(width = 1, colour = "white") + coord_polar("y") +
    labs(title = "Composición porcentual", x = NULL, y = NULL) +
    theme_minimal(base_size = 11.5) +
    theme(plot.background = element_rect(fill = "white", colour = NA),
          plot.title = element_text(colour = "#1a3a5c", face = "bold", size = 11.5),
          legend.title = element_blank(), legend.text = element_text(size = 9),
          legend.position = "right", axis.text = element_blank(), panel.grid = element_blank())
  print(p1); print(p2)
  invisible(NULL)
}

Nota de alcance y supuestos

Alcance. Los 31 análisis anteriores son estrictamente univariados: cada variable se describió por separado, con sus propias medidas y gráficos. No se incluyeron cruces de variables, medidas de asociación ni correlaciones, y tampoco técnicas multivariadas (componentes principales, agrupamiento) ni modelos de clasificación: todo eso corresponde a las entregas siguientes.

Supuestos y decisiones documentadas.

  1. Valores perdidos sin imputar. Las cinco variables con faltantes pertenecen al enriquecimiento (nombre, sentimiento y anotaciones de imagen). Se reportan y se describen solo con los casos válidos; la estrategia de imputación se definirá en la fase de modelamiento.
  2. Los faltantes no son aleatorios. El sentimiento falta en anuncios para los que no se recuperó la anotación de la API (no en relatos vacíos) y las variables de imagen faltan en los anuncios sin fotografía; conviene tratarlos con indicadores de ausencia.
  3. Conteos acumulados. total_labels, total_faces y total_colors acumulan las anotaciones de todas las fotografías del anuncio, no son promedios por imagen; su magnitud depende del número de fotos de la ficha.
  4. Códigos sin diccionario. Las razas, los colores y los estados se conservan como identificadores numéricos porque los archivos breed_labels.csv, color_labels.csv y state_labels.csv de la competencia no forman parte de la carpeta data/ del proyecto. Si se agregan, las tablas se pueden etiquetar sin cambiar el análisis.
  5. Atípicos conservados. No se eliminó ningún registro: los valores atípicos se documentaron con dos criterios y se interpretaron, porque en esta base corresponden a casos reales (animales mayores, anuncios de refugios, fichas muy detalladas).
  6. Forma de las distribuciones. La asimetría y la curtosis se calcularon con los estimadores muestrales tipo 2 de e1071 (la curtosis es exceso respecto de la normal).
  7. Reproducibilidad. Todo el análisis se ejecuta al construir el sitio con quarto render, a partir de data/petfinder_enriquecido_final.csv y del script scripts/01_dataset_enriquecido.R.

# --- formateo de cifras en español ---
fmt <- function(x, d = 2) {
  x <- as.numeric(x)
  if (all(abs(x - round(x)) < 1e-9)) {
    return(format(round(x), big.mark = ".", decimal.mark = ",", scientific = FALSE))
  }
  format(round(x, d), big.mark = ".", decimal.mark = ",", nsmall = d, scientific = FALSE)
}
ent <- function(x) format(as.integer(round(as.numeric(x))), big.mark = ".", decimal.mark = ",")
pct <- function(x, d = 1) gsub("\\.", ",", sprintf(paste0("%.", d, "f %%"), x))

# --- estadísticos ---
moda      <- function(x) { x <- x[!is.na(x)]; ux <- unique(x); ux[which.max(tabulate(match(x, ux)))] }
asimetria <- function(x) { x <- x[!is.na(x)]; n <- length(x); m <- mean(x); s <- sd(x); n/((n-1)*(n-2))*sum(((x-m)/s)^3) }
curtosis  <- function(x) { x <- x[!is.na(x)]; n <- length(x); m <- mean(x); s <- sd(x)
  n*(n+1)/((n-1)*(n-2)*(n-3))*sum(((x-m)/s)^4) - 3*(n-1)^2/((n-2)*(n-3)) }

esc_num <- function(x) {
  x <- as.numeric(x); v <- x[!is.na(x)]; q <- quantile(v, c(.25, .75)); iqr <- diff(q)
  z <- (v - mean(v)) / sd(v); li <- q[1] - 1.5*iqr; ls <- q[2] + 1.5*iqr
  list(n = length(v), na = sum(is.na(x)), media = mean(v), mediana = median(v), moda = as.numeric(moda(v)),
       min = min(v), max = max(v), q1 = q[1], q3 = q[2], iqr = iqr, rango = max(v) - min(v),
       var = var(v), sd = sd(v), cv = 100*sd(v)/mean(v), asim = asimetria(v), curt = curtosis(v),
       out_iqr = sum(v < li | v > ls), pct_out_iqr = 100*mean(v < li | v > ls),
       out_z = sum(abs(z) > 3), pct_out_z = 100*mean(abs(z) > 3), pct_ceros = 100*mean(v == 0),
       unicos = length(unique(v)))
}

esc_cat <- function(x) {
  v <- as.character(x); tb <- sort(table(v), decreasing = TRUE)
  list(n = length(v), niveles = length(tb), moda = names(tb)[1], n_moda = as.integer(tb[1]),
       pct_moda = 100*tb[1]/length(v), raras = sum(tb/length(v) < 0.01),
       pct_raras = 100*sum(tb[tb/length(v) < 0.01])/length(v),
       unicos = length(tb), unicidad = 100*length(tb)/length(v))
}

tabla_atipicos <- function(x, k = 3) {
  x <- as.numeric(x); v <- x[!is.na(x)]; q <- quantile(v, c(.25, .75)); iqr <- diff(q)
  z <- (v - mean(v))/sd(v); li <- q[1] - 1.5*iqr; ls <- q[2] + 1.5*iqr
  tibble::tibble(
    Criterio = c("Regla de 1,5 × IQR", paste0("Puntaje z (|z| > ", k, ")")),
    `Límite inferior` = c(fmt(li), fmt(mean(v) - k*sd(v))),
    `Límite superior` = c(fmt(ls), fmt(mean(v) + k*sd(v))),
    Atípicos = c(ent(sum(v < li | v > ls)), ent(sum(abs(z) > k))),
    `% del total` = c(pct(100*mean(v < li | v > ls)), pct(100*mean(abs(z) > k)))
  ) |> knitr::kable(align = c("l", "r", "r", "r", "r"))
}

tabla_frecuencias <- function(x, etiqueta = "Categoría", n_top = 10) {
  v <- as.character(x); tb <- sort(table(v), decreasing = TRUE); n <- length(tb); tot <- sum(tb)
  if (n > n_top) {
    tab <- tibble::tibble(
      Categoría = c(names(tb)[1:(n_top - 1)], sprintf("Otras (%d categorías)", n - n_top + 1)),
      Frecuencia = c(as.integer(tb[1:(n_top - 1)]), tot - sum(tb[1:(n_top - 1)])))
  } else {
    tab <- tibble::tibble(Categoría = names(tb), Frecuencia = as.integer(tb))
  }
  tab |>
    mutate(`Frecuencia relativa` = 100*Frecuencia/sum(Frecuencia),

}

### Nota de alcance y supuestos {#alcance}

**Alcance.** Los 31 análisis anteriores son estrictamente univariados: cada variable se describió
por separado, con sus propias medidas y gráficos. No se incluyeron cruces de variables, medidas de
asociación ni correlaciones, y tampoco técnicas multivariadas (componentes principales,
agrupamiento) ni modelos de clasificación: todo eso corresponde a las entregas siguientes.

**Supuestos y decisiones documentadas.**

1. **Valores perdidos sin imputar.** Las cinco variables con faltantes pertenecen al
   enriquecimiento (nombre, sentimiento y anotaciones de imagen). Se reportan y se describen solo
   con los casos válidos; la estrategia de imputación se definirá en la fase de modelamiento.
2. **Los faltantes no son aleatorios.** El sentimiento falta en anuncios para los que no se
   recuperó la anotación de la API (no en relatos vacíos) y las variables de imagen faltan en los
   anuncios sin fotografía; conviene tratarlos con indicadores de ausencia.
3. **Conteos acumulados.** `total_labels`, `total_faces` y `total_colors` acumulan las anotaciones
   de todas las fotografías del anuncio, no son promedios por imagen; su magnitud depende del
   número de fotos de la ficha.
4. **Códigos sin diccionario.** Las razas, los colores y los estados se conservan como
   identificadores numéricos porque los archivos `breed_labels.csv`, `color_labels.csv` y
   `state_labels.csv` de la competencia no forman parte de la carpeta `data/` del proyecto. Si se
   agregan, las tablas se pueden etiquetar sin cambiar el análisis.
5. **Atípicos conservados.** No se eliminó ningún registro: los valores atípicos se documentaron
   con dos criterios y se interpretaron, porque en esta base corresponden a casos reales
   (animales mayores, anuncios de refugios, fichas muy detalladas).
6. **Forma de las distribuciones.** La asimetría y la curtosis se calcularon con los estimadores
   muestrales tipo 2 de `e1071` (la curtosis es *exceso* respecto de la normal).
7. **Reproducibilidad.** Todo el análisis se ejecuta al construir el sitio con
